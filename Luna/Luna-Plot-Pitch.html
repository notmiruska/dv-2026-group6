#Select relevant reference-level variables
count_cols <- c("REKENEN_LAGER1F", "REKENEN_1F", "REKENEN_1S",
                "LV_LAGER1F", "LV_1F", "LV_2F")

# value for DUO's "<5" (1-4 pupils): the midpoint
lt5_value <- 2.5

replace_lt5 <- function(x) {
  x <- as.character(x)
  as.numeric(if_else(x == "<5", as.character(lt5_value), x))
}

#Convert columns to numeric, and turn values like <5 to NA
#Clean reference-level counts
ref_levels <- referentieniveaus |>
  mutate(across(all_of(count_cols), replace_lt5)) |> #used to turn <5 into NA, now into 2.5 to prevent false 100$ scores? For small schools, this quite greatly influenced averages for example. 
  group_by(INSTELLINGSCODE) |>
  summarise(
    GEMEENTENAAM = first(GEMEENTENAAM),
    INSTELLINGSNAAM_VESTIGING = first(INSTELLINGSNAAM_VESTIGING),
    across(all_of(count_cols), ~ sum(.x, na.rm = TRUE)), .groups = "drop") |>
  mutate(
    #total number of students for maths and for reading
    rekenen_n = REKENEN_LAGER1F + REKENEN_1F + REKENEN_1S,
    reading_n = LV_LAGER1F + LV_1F + LV_2F,
    #percentage of students for each level
    math_1F = 100 * (REKENEN_1F + REKENEN_1S) / rekenen_n,
    math_1S = 100 * REKENEN_1S / rekenen_n,
    reading_1F = 100 * (LV_1F + LV_2F) / reading_n,
    reading_2F = 100 * LV_2F / reading_n
    #DUO only count each pupil once, so to get the totals for reaching 1F in both reading and math, you must also count         the pupils that reached 1S and 2F
  ) |>
  filter(rekenen_n > 0, reading_n > 0)
    #filter out schools where no pupils are counted for maths and reading, this might not occur for Amstelveen, but is          necessary for calculating averages.

#calculate Schoolweging
