#Select relevant reference-level variables
count_cols <- c("REKENEN_LAGER1F", "REKENEN_1F", "REKENEN_1S",
                "LV_LAGER1F", "LV_1F", "LV_2F")

# value for DUO's "<5" (1-4 pupils): the midpoint
lt5_value <- 2.5

replace_lt5 <- function(x) {
  x <- as.character(x)
  as.numeric(if_else(x == "<5", as.character(lt5_value), x))
}

#Convert columns to numeric, and turn values like <5 to NA
#Clean reference-level counts
ref_levels <- referentieniveaus |>
  mutate(across(all_of(count_cols), replace_lt5)) |> #used to turn <5 into NA, now into 2.5 to prevent false 100$ scores? For small schools, this quite greatly influenced averages for example. 
  group_by(INSTELLINGSCODE) |>
  summarise(
    GEMEENTENAAM = first(GEMEENTENAAM),
    INSTELLINGSNAAM_VESTIGING = first(INSTELLINGSNAAM_VESTIGING),
    across(all_of(count_cols), ~ sum(.x, na.rm = TRUE)), .groups = "drop") |>
  mutate(
    #total number of students for maths and for reading
    rekenen_n = REKENEN_LAGER1F + REKENEN_1F + REKENEN_1S,
    reading_n = LV_LAGER1F + LV_1F + LV_2F,
    #percentage of students for each level
    math_1F = 100 * (REKENEN_1F + REKENEN_1S) / rekenen_n,
    math_1S = 100 * REKENEN_1S / rekenen_n,
    reading_1F = 100 * (LV_1F + LV_2F) / reading_n,
    reading_2F = 100 * LV_2F / reading_n
    #DUO only count each pupil once, so to get the totals for reaching 1F in both reading and math, you must also count         the pupils that reached 1S and 2F
  ) |>
  filter(rekenen_n > 0, reading_n > 0)
    #filter out schools where no pupils are counted for maths and reading, this might not occur for Amstelveen, but is          necessary for calculating averages.

#calculate Schoolweging
school_weights <- schoolweging |>
  mutate(INSTELLINGSCODE = str_remove(OVT, "\\|.*$")) |>
  group_by(INSTELLINGSCODE) |>
  summarise(schoolweging = weighted.mean(schoolweging, w = aantal_leerlingen),
    .groups = "drop")

#Combine reference-level results with schoolweging
school_data <- left_join(ref_levels, school_weights, by = "INSTELLINGSCODE")

#Amstelveen Gemeente
amstelveen_data <- filter(school_data, GEMEENTENAAM == "Amstelveen")

Create quadrant names

amstelveen_data <- amstelveen_data |>
  mutate(
    maths_above = math_1S >= national$math_1S,
    reading_above = reading_2F >= national$reading_2F,
    quadrant = case_when(
      maths_above & reading_above ~ "Above average in both",
      !maths_above & !reading_above ~ "Below average in both",
      TRUE ~ "Above average in one subject"
    )
  )

ggplot(data = amstelveen_data,
       mapping = aes(x = reading_2F, y = math_1S)) +
  geom_vline(xintercept = national$reading_2F,
             linetype = "dashed", color = "grey50") +
  geom_hline(yintercept = national$math_1S,
             linetype = "dashed", color = "grey50") +
  geom_label(data = reading_label, aes(x = x, y = y, label = label),
             hjust = 0, vjust = 1, size = 3, fontface = "italic",
             color = "grey40") +
  geom_label(data = maths_label, aes(x = x, y = y, label = label),
             hjust = 1, vjust = 0, size = 3, fontface = "italic",
             color = "grey40") +
  geom_point(aes(color = quadrant), size = 3) +
  geom_text_repel(aes(label = INSTELLINGSNAAM_VESTIGING), size = 3,
                  box.padding = 0.5, seed = 1, segment.color = NA) +
  scale_color_manual(
    values = c(
      # colours also visible for colour-blind readers!
      "Above average in both" = "#009E73",
      "Above average in one subject" = "#E69F00",
      "Below average in both" = "#B2182B"
    ),
    breaks = c(
      "Above average in both",
      "Above average in one subject",
      "Below average in both"
    )
  ) +
  labs(
    title = "Amstelveen schools compared to the national average",
    x = "% of pupils reaching reading level 2F",
    y = "% of pupils reaching maths level 1S",
    color = NULL
  ) +
  theme_minimal() +
